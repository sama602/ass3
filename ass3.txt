#include<stdio.h>
#include<semaphore.h>
#include<unistd.h>
#include<pthread.h>

#define ITEM 10
#define SIZE 5

sem_t empty;
sem_t full;
pthread_mutex_t mutex;

int buffer[SIZE];
int in=0;
int out=0;

void *producer(void *args){
    int items;
    for(int i=0;i<ITEM;i++){
        items=i;
        sem_wait(&empty);
        pthread_mutex_lock(&mutex);
        buffer[in]=items;
        printf("Producer Produced %d \n",items);
        in=(in+1)%SIZE;
        pthread_mutex_unlock(&mutex);
        sem_post(&full);
        sleep(1);
    }
    return NULL;
}
void *consumer(void *args){
    int items;
    for(int i=0;i<ITEM;i++){
        items=i;
        sem_wait(&full);
        pthread_mutex_lock(&mutex);
        items=buffer[out];
        printf("Consumer Consumed %d \n",items);
        out=(out+1)%SIZE;
        pthread_mutex_unlock(&mutex);
        sem_post(&empty);
        sleep(2);
    }
    return NULL;
}
int main(){
    pthread_t p,c;

    sem_init(&empty,0,SIZE);
    sem_init(&full,0,0);

    pthread_mutex_init(&mutex,NULL);

    pthread_create(&p,NULL,producer,NULL);
    pthread_create(&c,NULL,consumer,NULL);

    pthread_join(p,NULL);
    pthread_join(c,NULL);

    sem_destroy(&empty);
    sem_destroy(&full);

    pthread_mutex_destroy(&mutex);
    return 0;
}

gcc Assignment3.c -o Assignment3 -pthread
